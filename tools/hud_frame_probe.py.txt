class Surface(c.Structure):
    _fields_=[('data',c.c_void_p),('size',c.c_size_t),('width',c.c_uint32),('height',c.c_uint32),('stride',c.c_uint32),('bytes_per_pixel',c.c_uint8)]
class Color(c.Structure):
    _fields_=[('lookup_entry',c.c_uint32),('color',c.c_uint32),('half_color',c.c_uint32),('indices',c.c_uint16*3),('channels',c.c_uint8*3),('hue',c.c_uint8)]
class Preparation(c.Structure):
    _fields_=[('color',Color),('source_weights',c.c_void_p),('destination_weights',c.c_void_p),('source_words',c.c_void_p),('destination_words',c.c_void_p),('map_shift',c.c_int16),('cycle_enabled',c.c_uint8),('cycle',c.c_uint8)]
class Weights(c.Structure):
    _fields_=[('bytes',(c.c_uint8*256)*101),('words',(c.c_uint16*512)*101),('alpha',(c.c_uint8*256)*256),('inverse_alpha',(c.c_uint8*256)*256)]
class Tables(c.Structure):
    _fields_=[('lookup',c.POINTER(c.c_uint32)),('hue',c.POINTER(c.c_uint8)),('channels',c.c_uint8*20301),('auxiliary',c.c_uint32*255)]
class Bank(c.Structure):
    _fields_=[('lookup',c.c_uint32*13),('colors',c.c_uint32*13),('format_colors',c.c_uint32*12)]

def draw(bg,frame_index,item,x,y,scale=1):
    fr=Frame()
    assert native.vertrix_image_sprite_processed(archive,b'sprite/damier.spr',frame_index,masks,c.byref(fr))
    w,h=fr.image.width,fr.image.height
    actor=bytearray(0x15c)
    struct.pack_into('<H',actor,0xe,x&0xffff); struct.pack_into('<H',actor,0x16,y&0xffff); struct.pack_into('<H',actor,0x1e,0)
    actor[0x44]=scale; actor[0x8e]=1
    actor[0x93]=item[0]; actor[0x95]=item[1]; actor[0x4e]=item[2]
    struct.pack_into('<h',actor,0x48,struct.unpack_from('<h',item,4)[0])
    actor[0x52]=item[6]; actor[0x94]=item[7]; actor[0x8c]=item[8]; actor[0x80]=item[10]
    struct.pack_into('<I',actor,0xb4,0x801000); struct.pack_into('<I',actor,0x134,0x804000)
    struct.pack_into('<I',actor,0x13c,1); struct.pack_into('<I',actor,0x144,1)
    struct.pack_into('<I',actor,0xb0,0x5449fc); struct.pack_into('<I',actor,0xb8,1)
    prep=Preparation()
    amt=item[2]
    prep.source_weights=c.addressof(weights.bytes[amt]); prep.destination_weights=c.addressof(weights.bytes[100-amt])
    prep.source_words=c.addressof(weights.words[amt]); prep.destination_words=c.addressof(weights.words[100-amt])
    storage=c.create_string_buffer(bytes(actor),len(actor))
    surf=Surface(c.addressof(bg),len(bg),640,480,640*4,4)
    st=native.vertrix_actor_render_draw32(storage,len(actor),c.byref(fr),c.byref(surf),Clip(0,639,0,479),c.byref(weights),c.byref(bank),c.byref(tables),c.byref(prep))
    return st
